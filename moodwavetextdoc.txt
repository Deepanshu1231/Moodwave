// Grab all the playlist items (works on both watch page queues and playlist pages)
const items = document.querySelectorAll('ytd-playlist-panel-video-renderer, ytd-playlist-video-renderer');

const songs = Array.from(items).map(item => {
  // Find the title element inside the item
  const titleEl = item.querySelector('#video-title');
  // Clean up any weird spacing or newlines from the text
  const title = titleEl ? titleEl.textContent.replace(/\s+/g, ' ').trim() : 'Unknown Title';
  
  // Find the link to extract the ID
  const linkEl = item.querySelector('a[href*="watch?v="]');
  let id = null;
  if (linkEl) {
    try {
      id = new URL(linkEl.href, window.location.origin).searchParams.get('v');
    } catch (e) {
      id = null;
    }
  }
  
  return { id, title };
}).filter(song => song.id !== null);

// Remove duplicates in case YouTube loaded an item twice in the background
const uniqueSongs = [...new Map(songs.map(song => [song.id, song])).values()];
